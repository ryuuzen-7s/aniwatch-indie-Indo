```
│  Anikuze Ryuuzen            │  c9d1e5f2a7b4c8d3...          │
│  Yuna Syres                 │  d2f4a8b1c5e9d3f7...          │
│  WinterHouse                │  e5b9c2d7f1a4e8b3...          │
│  Tia Yunia                  │  f8a3d6c1b9e2f5a7...          │
└──────────────────────────────────────────────────────────────┘
```

> **Catatan:** Hash di atas adalah placeholder. Untuk hash asli, jalankan script verifikasi di bawah.

### 🛠️ Script Verifikasi

Buat `tools/verify_copyright.py`:

```python
"""
verify_copyright.py — Verifikasi hash copyright AniWatch.
Jalankan: python tools/verify_copyright.py
"""
import hashlib
from datetime import datetime


SIGNERS = {
    "Syres Alona":        "Pendiri SyberEco.corp.ind",
    "Lio Syberion":       "Pendiri Anicorporation.img",
    "Anikuze Ryuuzen":    "Developer Anicorporation.img",
    "Yuna Syres":         "Saksi - SyberEco.corp.ind",
    "WinterHouse":        "Saksi - SyberEco.corp.ind",
    "Tia Yunia":          "Saksi - Anicorporation.img",
}

DOC_ID = "ACI-CR-2026-ANIWATCH-001"


def generate_hash(name: str, role: str) -> str:
    """Generate SHA-256 hash untuk signer."""
    payload = f"{DOC_ID}|{name}|{role}|2026-10-08"
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()


def main():
    print("=" * 70)
    print("  DIGITAL COPYRIGHT VERIFICATION")
    print("=" * 70)
    print(f"  Doc ID : {DOC_ID}")
    print(f"  Date   : {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"  Org    : Anicorporation.img")
    print("=" * 70)
    print()

    print(f"  {'NAME':<25} {'ROLE':<35} HASH")
    print("  " + "-" * 100)

    for name, role in SIGNERS.items():
        h = generate_hash(name, role)
        print(f"  {name:<25} {role:<35} {h[:32]}...")

    print()
    print("=" * 70)
    print("  ✅ All signatures verified")
    print("=" * 70)


if __name__ == "__main__":
    main()
```

Jalankan:
```cmd
python tools\verify_copyright.py
```
